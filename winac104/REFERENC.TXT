【_wAlloc】       メモリブロックの確保     malloc同様にお使いください
---------------------------------------------------------------------
    構文: lptr = _wAlloc( size );

    引数: WORD       size;  要求するメモリブロックのサイズ(byte)
                            範囲:[ 1～65,524 ]
                            約2,000byte以下の場合に最高のパフォーマンス
                            を発揮します。

    戻値: LPVOID     lptr;  !=NULL: ＦＡＲポインタ(32bit)
　　　　　　　　　　　　　　==NULL: 確保失敗

    処理: 指定されたサイズのメモリブロックを確保し、そのＦＡＲポインタ
          を返します。
          確保されたメモリブロックは、０クリアされていません。
　　　　　ただし、デバッグモードでは常に０クリアされます。

    備考: _wInit()によりエラーハンドラを設定している場合には、戻値を確
    　　　認する必要はありません。万一確保できなかった場合には、エラー
    　　　ハンドラが呼び出されます。


=====================================================================
【_wCheck】         各種の診断をします
---------------------------------------------------------------------
    構文: iRet = _wCheck( mode );

    引数: int        mode;  =0: オーバーライトを診断する
                            =1: メモリリークについて診断します。
                                すべてのメモリブロックが解放されている
                                かを確認できます。

    戻値: int        iRet;  ==0: OK
                            !=0: ERROR

    処理: 各ページ内のヘッダとすべてのメモリブロックに付加されたチェッ
　　　　  クコードを調べます。
          オーバーライトしていそうな、行の直後に挿入すれば、バグの早
          期発見が可能です。

    備考: デバッグモード /D_AP_DEBUG の時だけ有効になります。
          リリースモードでの負荷はありません。


=====================================================================
【_wFree】          メモリブロックの解放
---------------------------------------------------------------------
    構文: _wFree( lptr );

    引数: LPVOID     lptr;  ＦＡＲポインタ(32bit)
                            _wAlloc, _wRealloc の戻値

    戻値: void

    処理: メモリブロックを解放します。
          解放されたメモリブロックは、未使用のメモリとして再利用され
          ます。


=====================================================================
【_wFreeAll】       メモリブロックの一斉高速解放
---------------------------------------------------------------------
    構文: _wFreeAll();

    引数: void

    戻値: void

    処理: すべてのメモリブロックを最短時間で解放します。
          解放されたメモリブロックは、未使用のメモリとして再利用され
          ます。

【_wInit】          ＷｉｎＡｌｌｏｃの初期化
---------------------------------------------------------------------
    構文: iRet = _wInit( lpCatchBuf, errFunc );

    引数: LPCATCHBUF lpCatchBuf;  Windows実行環境バッファへのポインタ
          int  (*errFunc)(int,char*);
                                  致命的エラー発生時に呼び出される関数
                                  へのポインタ
                                  =NULL: 内部エラールーチンに任せる

    戻値: int        iRet;  ==0: OK
                            !=0: ERROR

    処理: ＷｉｎＡｌｌｏｃの使用準備をします。

    備考: errFuncをセットすることにより、エラー発生時の処理ができます。
          また、_wAlloc,_wReallocの戻値をチェックする必要がなくなります。
          つまり、errFuncが呼び出されない限り、正しいメモリブロックが
          確保されたと考えられます。
          o errFunc()の例は、サンプルリスト "Wa_deb.c" をご参照下さい。
          o errFunc()の第１引数として与えられるエラー番号は以下のとおりです。
              継続可能なエラー --> return(0|1|2);
                 0: Out of memory.
                 1: 警告
                 2: Memory CHECK 警告
                 3: GlobalFree 失敗
              致命的エラー --> 強制終了
                50: 致命的エラー
                51: 引数エラー
                52: Memory CHECK エラー
                53: 初期化されていません
          o デバッグモードのときは、errFunc()の第２引数として、
          　詳細なメッセージが与えられます。

    注意: 最初に必ず呼び出して下さい。
          通常は、InitInstance() に挿入します。
          サンプルリスト "Walloc.c" をご参照下さい。

【_wRealloc】       メモリブロックのサイズ変更
---------------------------------------------------------------------
    構文: lpNew = _wRealloc( lpOld, size );

    引数: LPVOID    lpOld;  ＦＡＲポインタ(32bit)
          WORD       size;  >0: 新たなメモリブロックのサイズ(byte)

    戻値: LPVOID    lpNew;  !=NULL: 変更後のＦＡＲポインタ(32bit)
　　　　　　　　　　　　　　==NULL: 変更失敗

    処理: lpOldで示されるメモリブロックを、新たなサイズで確保されたメ
          モリブロックへ移動し、lpOldを破棄します。
          そして、新たなメモリブロックへのポインタが、lpNewとして返さ
          れます。
          メモリブロックの内容は、新たなメモリブロックへコピーされます
          が、小さくなった場合には、末尾部分より欠けることになります。
          また、大きくなった部分は、０クリアされません。
　　　　　ただし、デバッグモードでは常に０クリアされます。

    備考: _wInit()によりエラーハンドラを設定している場合には、戻値を確
    　　　認する必要はありません。万一変更できなかった場合には、エラー
    　　　ハンドラが呼び出されます。

#include "windows.h"        /* すべてのWindowsアプリケーションに必要です */
#include "WinAlloc.h"       /* WinAlloc Lib. */

/*** WinAlloc< 0 > Windows実行環境バッファ宣言 ***/
CATCHBUF  CatchBuf;     /* _wInitに渡すため */

int PASCAL
WinMain(hInstance, hPrevInstance, lpCmdLine, nCmdShow)
{
    /*** WinAlloc< 1 > Windows動作モードを調べる ***/
    if( !(GetWinFlags() & WF_PMODE) ) /* プロテクトモードではない */
        return FALSE;  /* Error */

    /*** WinAlloc< 2 > Windows実行環境を得る ***/
    if( Catch( (LPCATCHBUF)&CatchBuf )==(-1) ) return FALSE;  /* Error */

    if(!hPrevInstance)

    if (!InitInstance(hInstance, nCmdShow)) return FALSE;

    /* メッセージ・ループ */
}

BOOL
InitInstance(hInstance, nCmdShow)
{
    hWnd = CreateWindow( ,,,,, );
    ShowWindow(hWnd, nCmdShow);
    UpdateWindow(hWnd);
    
    /*** WinAlloc< 3 > WinAllocを初期化する ***/
    if( !_wInit( (LPCATCHBUF)&CatchBuf, errHandler ) ) return FALSE;
    
    return TRUE;
}

long FAR PASCAL
MainWndProc(hWnd, message, wParam, lParam)
{
    switch(message)
    {
        case WM_DESTROY:         /* メッセージ: ウィンドウ廃棄       */
            /*** WinAlloc< 4 > WinAllocを終了する ***/
            _wTerm();
            PostQuitMessage(0);
            break;
        default:  /* 未処理ならそのメッセージを引き渡します */
            return DefWindowProc(hWnd, message, wParam, lParam);
    }
    return NULL;
}
