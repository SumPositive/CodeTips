メモリ　　　　　：プロテクトメモリ２Ｍ

#include <vm_dos.h>
#include <string.h>
#include <dos.h>
#include <graph.h>
main()
{
    char *pz;
    VMHAND vh;
    int (*errFUNC)(int,char*);
    
    graphset();
    _moveto(0,50);_lineto(639,50);
    
    vm_init(0,VM_ALLDOS,VM_EMS,2,errFUNC);
    vh=vm_alloc(2000);
    
    pz=(char *)vm_load(vh,VM_DIRTY);
    _getimage(0,0,300,100,pz);   /* 4+(((300+1)*4+7)/8)*(100+1)=15293 */
    _clearscreen(_GCLEARGRAPH);
    getch();
    pz=(char f*)vm_load(vh,VM_CLEAN);
    _putimage(0,0,pz,_GPSET);
    
    vm_free(vh);
    vm_term();
}

　  vm_init(0,VM_ALLDOS,VM_EMS, ８, errFUNC); として下さい。
　  それと、errFUNC()は、どこかに記述してあるのでしょうか？

Ａ．これは、デバッグモードを評価して頂く絶好のチャンスです。
  　コンパイラオプション /D_AP_DEBUG を付けてコンパイルして下さい。

  　オーバーライトが報告されるはずです。
  　このオーバーライトによって、エディタが動作不良を起こしたのでしょう。

  _getimage(0,0,639,50,pz1); これに必要なメモリは、少なくとも 1600バ
  イトより多いはずです。
  _imagesize();を用いれば、正確なサイズが解ります。

　ただし、ページ１６Ｋ や 1600バイトをアロケートするのはお薦めできません。
　ＶＭＭ＿Ｄｏｓは、ページやブロックのサイズが小さいほど効率が良くなります。
　ブロックのサイズには限りが有りますが、数は実質無限大です。

　例えば、このような方法はどうでしょうか。

    #define YSTEP  5
    
    WORD  wSize;
    int   y;
    char far *pz;
    VMHAND  vhTop, vh, *pvh;
    
    /* SAVE */
    wSize = sizeof(VMHAND) + (WORD)_imagesize(0,0, 639,YSTEP);
    for( vhTop=VM_NULL, y=0 ; y<400 ; y += YSTEP )
    {
        vh = vm_alloc( wSize );
        pz = (char far *)vm_load(vh,VM_DIRTY);
        
        pvh = (VMHAND *)pz;
        *pvh = vhTop;    /* リンク・ハンドルを書き込む */
        
        _getimage(0,y,639,y+YSTEP, (char _huge *)(pvh+1) );
        
        vhTop = vh;
    }
    
    getch();
    _clearscreen(_GCLEARGRAPH);
    getch();
    
    /* LOAD */
    wSize = sizeof(VMHAND) + (WORD)_imagesize(0,0, 639,YSTEP); /* 同じ */
    for( vh=vhTop, y=400-YSTEP ; 0<=y ; y -= YSTEP )  /* 逆になっている */
    {
        pz = (char far *)vm_load(vh,VM_CLEAN);
        
        pvh = (VMHAND *)pz;
        vh = *pvh;    /* 次のハンドルです */
        
        _putimage(0,y, (char _huge *)(pvh+1), _GPSET);
    }

　直接メールに書き込んだために、動作確認はしていません。
　参考程度にして下さい。

　最後に、/*本当にメモリーを食ってないか調べる*/ についてですが、
ユーザーズ・ガイドにある、[動作原理]に簡単に説明しているように、使用中の
ページは、コンベンショナル・メモリ（ヒープ）に存在します。そうしないとポ
インタ参照ができません。
　また、初期化で vm_init( ,VM_ALLDOS,,, );を指定すれば、すべてのヒープ
を使用許可したことになります。
　例えば、vm_init( , 64,,, );にすれば、ＶＭＭは、６４Ｋのヒープだけを
使ってやりくりします。
　使用許可ヒープサイズは、ページサイズよりも十分大きくしたほうが高効率です。
　ただし、ＶＭＭ＿Ｄｏｓは、ヒープが少ないほど Microsoft C7 付属の _v関数を
超越する性能を発揮します。

---------------------------------------------------------------------
Ａ．最初の０は、ＶＭＭに与える最小のヒープメモリ容量を指示します。
    ０ということは、もし、ヒープメモリに空きが全くなくても vm_init()は
    正常にﾘﾀｰﾝしますが、最初のアロケート(vm_allocなど)でエラーが発生し
    ます。
    ほとんどの場合０で問題ありません。

    次の０は、ＶＭＭに使用許可するヒープメモリ容量を指示します。
    ０は、現ﾊﾞｰｼﾞｮﾝでは、VM_ALLDOS の定義値と同じです。
    つまり、全てのヒープメモリを利用します。
